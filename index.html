<script>

  const key = document.getElementById('key');

  const door = document.getElementById('door');

  const keyInLock = document.getElementById('keyInLock');

  const darkRoom = document.getElementById('darkRoom');

  const light = document.getElementById('light');

  const hint = document.getElementById('hint');

  const victory = document.getElementById('victory');

  const extraKeys = document.querySelectorAll('.extra-key');

  

  let keyInDoor = false;

  let doorOpen = false;

  

  key.addEventListener('click', () => {

    if (keyInDoor || doorOpen) return;

    

    key.classList.add('taken');

    

    setTimeout(() => {

      keyInLock.classList.add('visible');

      keyInDoor = true;

      hint.textContent = '🔑 Ключ вставлен в скважину! Теперь нажми на дверь.';

    }, 400);

  });

  

  extraKeys.forEach(key => {

    key.addEventListener('click', () => {

      if (key.classList.contains('cloud-vanish')) return;

      key.classList.add('cloud-vanish');

      setTimeout(() => {

        key.style.pointerEvents = 'none';

      }, 600);

    });

  });

  

  door.addEventListener('click', () => {

    if (doorOpen) return;

    

    if (!keyInDoor) {

      door.classList.remove('shake');

      void door.offsetWidth;

      door.classList.add('shake');

      hint.textContent = '🔒 Дверь заперта! Нужен ключ...';

      setTimeout(() => door.classList.remove('shake'), 600);

      return;

    }

    

    doorOpen = true;

    door.classList.add('open');

    darkRoom.classList.add('visible');

    light.classList.add('on');

    hint.textContent = '🎉 Дверь открыта!';

    

    setTimeout(() => {

      victory.classList.add('show');

    }, 800);

  });

  

  function resetGame() {

    keyInDoor = false;

    doorOpen = false;

    key.classList.remove('taken');

    keyInLock.classList.remove('visible');

    door.classList.remove('open', 'shake');

    darkRoom.classList.remove('visible');

    light.classList.remove('on');

    victory.classList.remove('show');

    extraKeys.forEach(k => {

      k.classList.remove('cloud-vanish');

      k.style.pointerEvents = 'auto';

    });

    hint.textContent = 'Найди нужный ключ и открой дверь!';

  }

</script>
